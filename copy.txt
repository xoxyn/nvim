## lua/builds_cpp.lua

local M = {}

-- ── helpers ──────────────────────────────────────────────────
-- cmake-tools is cwd-based, so detect the same way
local function is_cmake_project()
    return vim.uv.fs_stat(vim.fn.getcwd() .. "/CMakeLists.txt") ~= nil
end

local function has_main(path)
    for _, line in ipairs(vim.fn.readfile(path)) do
        if line:match("%f[%w_]main%s*%(") then
            return true
        end
    end
    return false
end

-- ── scratch fallback (no CMakeLists.txt) ─────────────────────
local function run_scratch(ft, args)
    local file = vim.fn.expand("%:p")
    local dir = vim.fn.expand("%:p:h")
    local out = "/tmp/" .. vim.fn.expand("%:t:r")
    local ext = (ft == "cpp") and "cpp" or "c"

    local compiler, std = "gcc", "-std=c17"
    if ft == "cpp" then
        compiler, std = "g++", "-std=c++17"
    end

    -- current file first, then sibling sources that don't define main()
    local sources = { file }
    for _, f in ipairs(vim.fn.glob(dir .. "/*." .. ext, false, true)) do
        if f ~= file and not has_main(f) then
            table.insert(sources, f)
        end
    end

    local cmd = string.format(
        "%s %s -Wall -Wextra -Wpedantic -I%s -I%s -I%s %s -o %s && %s%s",
        compiler,
        std,
        vim.fn.shellescape(dir),
        vim.fn.shellescape(dir .. "/include"),
        vim.fn.shellescape(dir .. "/../include"),
        table.concat(vim.tbl_map(vim.fn.shellescape, sources), " "),
        vim.fn.shellescape(out),
        vim.fn.shellescape(out),
        args ~= "" and (" " .. args) or ""
    )
    vim.cmd("terminal " .. cmd)
end

-- ── dispatcher ───────────────────────────────────────────────
function M.run(with_args)
    local ft = vim.bo.filetype
    if not vim.tbl_contains({ "c", "cpp", "cmake" }, ft) then
        vim.notify("Not a C/C++ file", vim.log.levels.WARN)
        return
    end

    if is_cmake_project() then
        if with_args then
            -- opens an input popup; args are stored per launch target
            vim.cmd("CMakeLaunchArgs")
        else
            vim.cmd("CMakeRun") -- builds the launch target, then runs it
        end
    elseif ft ~= "cmake" then
        run_scratch(ft, with_args and vim.fn.input("Arguments: ") or "")
    end
end

-- Keymaps for Rust
-- Run without arguments
keymap("n", "<leader>rs", function()
    local filetype = vim.bo.filetype

    if filetype == 'rust' then
        local root_dir = vim.fn.getcwd()
        local cargo_toml = root_dir .. '/Cargo.toml'

        if vim.fn.filereadable(cargo_toml) == 1 then
            -- Cargo project: build and run
            vim.cmd('terminal cargo run')
        else
            -- Single file fallback via rustc
            local filename = vim.fn.expand('%')
            local output = '/tmp/' .. vim.fn.expand('%:t:r')
            vim.cmd('terminal rustc ' .. filename .. ' -o ' .. output .. ' && ' .. output)
        end
    else
        print('Not a Rust file')
    end
end, { desc = "Run Rust program or Cargo project" })

-- Run with arguments
keymap("n", "<leader>ra", function()
    local filetype = vim.bo.filetype

    if filetype == 'rust' then
        local root_dir = vim.fn.getcwd()
        local cargo_toml = root_dir .. '/Cargo.toml'

        local args = vim.fn.input('Arguments: ')

        if vim.fn.filereadable(cargo_toml) == 1 then
            -- Cargo project: pass args after '--'
            vim.cmd('terminal cargo run -- ' .. args)
        else
            -- Single file fallback via rustc
            local filename = vim.fn.expand('%')
            local output = '/tmp/' .. vim.fn.expand('%:t:r')
            vim.cmd('terminal rustc ' .. filename .. ' -o ' .. output .. ' && ' .. output .. ' ' .. args)
        end
    else
        print('Not a Rust file')
    end
end, { desc = "Run Rust program with arguments" })

local lsp = vim.lsp
vim.api.nvim_create_autocmd('LspAttach', {
    callback = function(args)
        local buf = args.buf
        local k = function(mode, lhs, rhs) vim.keymap.set(mode, lhs, rhs, { buffer = buf }) end
        k('n', 'gd', lsp.buf.definition)
        k('n', 'gD', lsp.buf.declaration)
        k('n', 'gr', lsp.buf.references)
        k('n', 'gi', lsp.buf.implementation)
        k('n', 'K', lsp.buf.hover)
        k('n', '<leader>rn', lsp.buf.rename)
        k('n', '<leader>ca', lsp.buf.code_action)
        -- Use conform for formatting (falls back to LSP if no formatter configured)
        k('n', '<leader>f', function()
            require('conform').format({ async = true, lsp_fallback = true })
        end)

        -- Removed this for Digraphs
        -- k('i', '<C-k>', function()
        --     require('blink.cmp').show()
        -- end)

        local client = vim.lsp.get_client_by_id(args.data.client_id)
        if client and client:supports_method('textDocument/semanticTokens') then
            vim.lsp.semantic_tokens.enable(true, { bufnr = buf })
        end
    end,
})

local function start(server, config)
    vim.api.nvim_create_autocmd('FileType', {
        pattern = config.filetypes,
        callback = function()
            -- Merge blink.cmp capabilities so completions work across all servers
            local capabilities = require('blink.cmp').get_lsp_capabilities()
            vim.lsp.start(vim.tbl_extend('keep', config, {
                name = server,
                root_dir = vim.fs.root(0, config.root_markers or { '.git' }),
                capabilities = capabilities,
            }))
        end,
    })
end
